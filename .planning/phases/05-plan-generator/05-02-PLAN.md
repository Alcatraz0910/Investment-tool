    ```typescript
    import { Decimal } from 'decimal.js'
    import type { AssetCategory } from '@/types'
    import type { BlendedStrategy } from '@/lib/strategy/blender'

    // HoldingWithFillTicker: extends Holding with isFillTicker
    // (Holding already has isFillTicker from 05-01, so this is just the Holding type)
    export type HoldingWithFillTicker = {
      id: string
      userId: string
      ticker: string
      category: AssetCategory
      currentValue: Decimal   // accepts Decimal from Holding type
      isFillTicker: boolean
    }

    export interface BuyListItem {
      ticker: string
      category: AssetCategory
      amountGbp: Decimal
      allocationGapPct: number
      rationale: string
    }

    export interface GapRow {
      category: AssetCategory
      targetPct: number
      currentPct: number
      gapPct: number
      reason: 'no-holdings' | 'no-fill-ticker'
    }

    export type PlanResult =
      | { type: 'no-strategy' }
      | { type: 'no-fill-tickers'; gapRows: GapRow[] }
      | { type: 'buy-list'; items: BuyListItem[]; gapRows: GapRow[]; isaWarning: boolean; effectiveBudget: Decimal }

    export function generatePlan(
      holdings: HoldingWithFillTicker[],
      budgetGbp: number,
      strategy: BlendedStrategy | null,
      isaRemaining: number,
    ): PlanResult
    ```

    Algorithm (all arithmetic via Decimal):
    1. Guard: if !strategy || Object.keys(strategy.unified).length === 0 → return { type: 'no-strategy' }
    2. Wrap: budget = new Decimal(budgetGbp), isaRem = new Decimal(isaRemaining)
    3. effectiveBudget = Decimal.min(budget, isaRem)
    4. isaWarning = budget.greaterThan(isaRem)
    5. portfolioTotal = holdings.reduce((s, h) => s.plus(h.currentValue), new Decimal(0))
    6. totalAfter = portfolioTotal.plus(effectiveBudget)
    7. For each category in strategy.unified with positive allocation:
       a. targetValue = totalAfter.mul(new Decimal(strategy.unified[cat])).div(100)
       b. currentValue = sum of holdings where category matches (Decimal sum)
       c. gap = targetValue.minus(currentValue)
       d. if gap <= 0: skip entirely (at or above target)
       e. fillHolder = holdings.find(h => h.category === cat && h.isFillTicker)
       f. if !fillHolder && holdings.some(h => h.category === cat): push GapRow(reason='no-fill-ticker'), continue
       g. if !fillHolder && !holdings.some(h => h.category === cat): push GapRow(reason='no-holdings'), continue
       h. if fillHolder: push { category, gap, fillHolder } to pendingItems
    8. If pendingItems is empty AND gapRows.length > 0 AND no fill tickers anywhere:
       return { type: 'no-fill-tickers', gapRows }
    9. If pendingItems is empty: return { type: 'buy-list', items: [], gapRows, isaWarning, effectiveBudget }
    10. totalGap = sum of pending gaps (Decimal)
    11. rawItems = pendingItems.map(p => rawAmount = effectiveBudget.mul(p.gap).div(totalGap))
    12. roundedItems = rawItems.map(r => r.toDecimalPlaces(2, Decimal.ROUND_HALF_UP))
    13. rawSum = roundedItems.reduce sum
    14. diff = effectiveBudget.minus(rawSum) — last item absorbs diff
    15. roundedItems[last] = roundedItems[last].plus(diff)
    16. Build BuyListItem per item:
        - ticker = fillHolder.ticker
        - category = p.category
        - amountGbp = roundedItems[i]
        - allocationGapPct = p.gap.div(totalAfter).mul(100).toDecimalPlaces(2).toNumber()
        - rationale = `${p.category} is ${currentPct.toFixed(0)}% of portfolio vs ${targetPct.toFixed(0)}% target — adding to ${ticker} closes ${allocationGapPct.toFixed(1)}% of the gap.`
          (MUST NOT contain 'advice', 'recommend', 'suggest')
    17. return { type: 'buy-list', items, gapRows, isaWarning, effectiveBudget }

    CRITICAL: generator.ts must have ZERO imports from @/lib/supabase/*, next/headers, or any server-only module. Only: decimal.js, @/types, @/lib/strategy/blender (type only).
