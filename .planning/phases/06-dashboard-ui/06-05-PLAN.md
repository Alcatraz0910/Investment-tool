```tsx
/**
 * UI-03: computeRoadmap pure-function tests.
 */
import { describe, it, expect } from 'vitest'
import { Decimal } from 'decimal.js'
import { computeRoadmap } from '@/lib/plan/roadmap'
import type { HoldingWithFillTicker } from '@/lib/plan/generator'
import type { BlendedStrategy } from '@/lib/strategy/blender'

const baseHolding: HoldingWithFillTicker = {
  id: 'h1',
  userId: 'u1',
  ticker: 'VUSA',
  category: 'Index Funds',
  quantity: new Decimal(10),
  currentValue: new Decimal(500),
  isFillTicker: true,
  createdAt: new Date(),
  updatedAt: new Date(),
}

describe('computeRoadmap (UI-03)', () => {
  it('returns [] when blend is null', () => {
    expect(computeRoadmap([baseHolding], null, 500, '2025-26')).toEqual([])
  })

  it('returns [] when blend.unified is empty', () => {
    const emptyBlend: BlendedStrategy = { unified: {}, influence: {} }
    expect(computeRoadmap([baseHolding], emptyBlend, 500, '2025-26')).toEqual([])
  })

  it('returns at least 1 point when tax year has remaining months', () => {
    const blend: BlendedStrategy = {
      unified: { 'Index Funds': 100 },
      influence: {},
    }
    const points = computeRoadmap([baseHolding], blend, 500, '2025-26')
    expect(points.length).toBeGreaterThanOrEqual(1)
  })

  it('month labels are formatted as "Mon YY"', () => {
    const blend: BlendedStrategy = {
      unified: { 'Index Funds': 100 },
      influence: {},
    }
    const points = computeRoadmap([baseHolding], blend, 500, '2025-26')
    // Each label should be e.g. "May 26" — 3 chars + space + 2 chars
    expect(points[0].month).toMatch(/^[A-Z][a-z]{2} \d{2}$/)
  })

  it('output values are plain numbers (not Decimal objects)', () => {
    const blend: BlendedStrategy = {
      unified: { 'Index Funds': 100 },
      influence: {},
    }
    const points = computeRoadmap([baseHolding], blend, 500, '2025-26')
    expect(typeof points[0].currentPath).toBe('number')
    expect(typeof points[0].creatorVision).toBe('number')
  })

  it('divergence: 0% Stocks current vs 60% Stocks blend → creatorVision grows faster than currentPath', () => {
    // User has ONLY Index Funds — no Stocks holdings
    const indexFundsHolding: HoldingWithFillTicker = {
      ...baseHolding,
      category: 'Index Funds',
      currentValue: new Decimal(1000),
    }
    const blend: BlendedStrategy = {
      unified: { 'Stocks': 60, 'Index Funds': 40 },
      influence: {},
    }
    const points = computeRoadmap([indexFundsHolding], blend, 500, '2025-26')

    // Need at least 2 points to compare month 0 vs month 1
    expect(points.length).toBeGreaterThanOrEqual(2)

    // Month 0: both start at same base (holdings in Stocks = 0)
    expect(points[0].currentPath).toBe(points[0].creatorVision)

    // Month 1: creatorVision grows by budget * 60% = 300; currentPath stays flat (0% Stocks)
    expect(points[1].creatorVision).toBeGreaterThan(points[1].currentPath)
  })
})
```
